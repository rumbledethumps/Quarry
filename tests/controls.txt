# Every control of milestone 1. Run from the project folder:
#   tools/rp6502-emu build/llvm-mos/debug/quarry.rp6502 --script tests/controls.txt --seed 1 --mute
#
# The paddle starts centered at x 148 and moves 2 px a step. Its left piece
# is the sprite at XRAM $0292. A move must reach 160 ($A0) or 136 ($88),
# and the screen must stop changing once the control is let go. Each test
# starts from the same saved state, with the ball on the paddle.

# Enter held from boot, as when it loaded the game, does not launch.
press enter
wait "stage start" 60
run 30
release enter
run 5
expect-not "launch"

run 2
peek xram:$0292 $94 $00
pad 0 connect
pad 1 connect
run 2
state save "build/controls.state"

# Num Lock is on when the emulator starts.
peek xram:$0000 $03

# d moves right, Num Lock on
state load "build/controls.state"
press d
wait xram:$0292 $A0 30
release d
run 3
mark
run 10
expect-same

# a moves left, Num Lock on
state load "build/controls.state"
press a
wait xram:$0292 $88 30
release a
run 3
mark
run 10
expect-same

# right moves right, Num Lock on
state load "build/controls.state"
press right
wait xram:$0292 $A0 30
release right
run 3
mark
run 10
expect-same
